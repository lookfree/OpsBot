# ZWD-OpsBot 数据库设计器功能规格文档

> 版本：v1.0
> 创建日期：2025-12-21
> 参考项目：DrawDB (https://github.com/drawdb-io/drawdb)
> 模块目标：可视化数据库 ER 图设计与 DDL 生成

---

## 1. 功能概述

### 1.1 目标

在 ZWD-OpsBot 中集成数据库设计功能，提供：
- 可视化 ER 图编辑器
- 多数据库方言 DDL 生成
- 与现有数据库管理模块的无缝集成

### 1.2 支持的数据库

| 数据库 | 状态 | 优先级 |
|--------|------|--------|
| MySQL | 已完成（基础连接） | P0 |
| PostgreSQL | 已完成（基础连接） | P0 |
| MariaDB | 待开发 | P1 |
| SQLite | 待开发 | P1 |
| SQL Server (MSSQL) | 待开发 | P2 |
| Oracle | 待开发 | P2 |

### 1.3 核心功能

1. **ER 图可视化编辑**
   - 表设计（字段、类型、约束）
   - 关系设计（外键、基数）
   - 索引管理
   - 注释与分组区域

2. **DDL 生成**
   - 根据 ER 图生成目标数据库的 CREATE TABLE 语句
   - 支持导出为 SQL 文件

3. **逆向工程**
   - 从现有数据库连接导入表结构
   - 自动生成 ER 图

## 3. 数据模型设计

### 3.1 核心数据结构

参考 DrawDB 的 JSON Schema 设计，定义以下核心数据结构：

```typescript
// front/components/database/designer/types.ts

// 支持的数据库类型
export type DatabaseDialect =
  | 'mysql'
  | 'postgresql'
  | 'mariadb'
  | 'sqlite'
  | 'mssql'
  | 'oracle';

// 外键约束动作
export type ConstraintAction =
  | 'NO ACTION'
  | 'RESTRICT'
  | 'CASCADE'
  | 'SET NULL'
  | 'SET DEFAULT';

// 关系基数
export type Cardinality =
  | 'one_to_one'
  | 'one_to_many'
  | 'many_to_one';

// 字段定义
export interface TableField {
  id: string;                    // UUID
  name: string;                  // 字段名
  type: string;                  // 数据类型 (如 VARCHAR, INT)
  size?: string | number;        // 类型大小 (如 VARCHAR(255))
  default: string;               // 默认值
  check: string;                 // CHECK 约束表达式
  primary: boolean;              // 是否主键
  unique: boolean;               // 是否唯一
  notNull: boolean;              // 是否非空
  increment: boolean;            // 是否自增
  unsigned?: boolean;            // 是否无符号 (MySQL/MariaDB)
  isArray?: boolean;             // 是否数组 (PostgreSQL)
  comment: string;               // 字段注释
  values?: string[];             // ENUM/SET 值列表
}

// 索引定义
export interface TableIndex {
  id: string;                    // UUID
  name: string;                  // 索引名
  unique: boolean;               // 是否唯一索引
  fields: string[];              // 包含的字段名列表
}

// 表定义
export interface TableNode {
  id: string;                    // UUID
  name: string;                  // 表名
  x: number;                     // 画布 X 坐标
  y: number;                     // 画布 Y 坐标
  fields: TableField[];          // 字段列表
  indices: TableIndex[];         // 索引列表
  comment: string;               // 表注释
  color: string;                 // 表头颜色 (如 #175e7a)
  locked?: boolean;              // 是否锁定位置
  hidden?: boolean;              // 是否隐藏
  inherits?: string[];           // 继承表 (PostgreSQL)
}

// 关系定义
export interface Relationship {
  id: string;                    // UUID
  name: string;                  // 关系名（外键约束名）
  startTableId: string;          // 起始表 ID
  startFieldId: string;          // 起始字段 ID
  endTableId: string;            // 目标表 ID
  endFieldId: string;            // 目标字段 ID
  cardinality: Cardinality;      // 基数关系
  updateConstraint: ConstraintAction;  // ON UPDATE 动作
  deleteConstraint: ConstraintAction;  // ON DELETE 动作
}

// 注释节点
export interface NoteNode {
  id: string;                    // UUID
  x: number;                     // 画布 X 坐标
  y: number;                     // 画布 Y 坐标
  title: string;                 // 标题
  content: string;               // 内容 (支持 Markdown)
  color: string;                 // 背景颜色
  height: number;                // 高度
  locked?: boolean;              // 是否锁定
}

// 分组区域
export interface AreaNode {
  id: string;                    // UUID
  name: string;                  // 区域名称
  x: number;                     // X 坐标
  y: number;                     // Y 坐标
  width: number;                 // 宽度
  height: number;                // 高度
  color: string;                 // 边框/背景颜色
  locked?: boolean;              // 是否锁定
}

// 自定义类型 (PostgreSQL)
export interface CustomType {
  id: string;
  name: string;
  fields: {
    id: string;
    name: string;
    type: string;
    values?: string[];           // ENUM 值
  }[];
  comment: string;
}

// 枚举类型 (PostgreSQL)
export interface EnumType {
  id: string;
  name: string;
  values: string[];
}

// 完整图表定义
export interface DiagramData {
  title: string;                 // 图表标题
  database: DatabaseDialect;     // 目标数据库
  tables: TableNode[];           // 表列表
  relationships: Relationship[]; // 关系列表
  notes: NoteNode[];             // 注释列表
  areas: AreaNode[];             // 分组区域列表
  types?: CustomType[];          // 自定义类型 (PostgreSQL)
  enums?: EnumType[];            // 枚举类型 (PostgreSQL)
}

// DDB 文件格式（包含元数据）
export interface DiagramFile {
  author: string;
  project: string;
  title: string;
  date: string;
  version: string;
  data: DiagramData;
}
```

### 3.2 状态管理

```typescript
// front/stores/diagramStore.ts

import { create } from 'zustand';
import { immer } from 'zustand/middleware/immer';
import {
  DiagramData,
  TableNode,
  Relationship,
  NoteNode,
  AreaNode,
  DatabaseDialect
} from '../components/database/designer/types';

interface DiagramState {
  // 图表数据
  diagram: DiagramData;

  // 选中状态
  selectedTableId: string | null;
  selectedRelationshipId: string | null;
  selectedNoteId: string | null;
  selectedAreaId: string | null;

  // 编辑状态
  isDirty: boolean;

  // 历史记录 (Undo/Redo)
  history: DiagramData[];
  historyIndex: number;

  // Actions
  setDatabase: (database: DatabaseDialect) => void;

  // 表操作
  addTable: (table: TableNode) => void;
  updateTable: (id: string, updates: Partial<TableNode>) => void;
  deleteTable: (id: string) => void;
  moveTable: (id: string, x: number, y: number) => void;

  // 字段操作
  addField: (tableId: string, field: TableField) => void;
  updateField: (tableId: string, fieldId: string, updates: Partial<TableField>) => void;
  deleteField: (tableId: string, fieldId: string) => void;
  reorderFields: (tableId: string, fieldIds: string[]) => void;

  // 索引操作
  addIndex: (tableId: string, index: TableIndex) => void;
  updateIndex: (tableId: string, indexId: string, updates: Partial<TableIndex>) => void;
  deleteIndex: (tableId: string, indexId: string) => void;

  // 关系操作
  addRelationship: (relationship: Relationship) => void;
  updateRelationship: (id: string, updates: Partial<Relationship>) => void;
  deleteRelationship: (id: string) => void;

  // 注释操作
  addNote: (note: NoteNode) => void;
  updateNote: (id: string, updates: Partial<NoteNode>) => void;
  deleteNote: (id: string) => void;

  // 区域操作
  addArea: (area: AreaNode) => void;
  updateArea: (id: string, updates: Partial<AreaNode>) => void;
  deleteArea: (id: string) => void;

  // 选择操作
  selectTable: (id: string | null) => void;
  selectRelationship: (id: string | null) => void;
  selectNote: (id: string | null) => void;
  selectArea: (id: string | null) => void;
  clearSelection: () => void;

  // 历史操作
  undo: () => void;
  redo: () => void;
  saveToHistory: () => void;

  // 导入/导出
  loadDiagram: (data: DiagramData) => void;
  exportDiagram: () => DiagramData;

  // 清空
  reset: () => void;
}

const initialDiagram: DiagramData = {
  title: 'Untitled Diagram',
  database: 'mysql',
  tables: [],
  relationships: [],
  notes: [],
  areas: [],
  types: [],
  enums: [],
};

export const useDiagramStore = create<DiagramState>()(
  immer((set, get) => ({
    diagram: initialDiagram,
    selectedTableId: null,
    selectedRelationshipId: null,
    selectedNoteId: null,
    selectedAreaId: null,
    isDirty: false,
    history: [initialDiagram],
    historyIndex: 0,

    setDatabase: (database) => set((state) => {
      state.diagram.database = database;
      state.isDirty = true;
    }),

    addTable: (table) => set((state) => {
      state.diagram.tables.push(table);
      state.isDirty = true;
    }),

    updateTable: (id, updates) => set((state) => {
      const table = state.diagram.tables.find(t => t.id === id);
      if (table) {
        Object.assign(table, updates);
        state.isDirty = true;
      }
    }),

    deleteTable: (id) => set((state) => {
      state.diagram.tables = state.diagram.tables.filter(t => t.id !== id);
      // 同时删除相关的关系
      state.diagram.relationships = state.diagram.relationships.filter(
        r => r.startTableId !== id && r.endTableId !== id
      );
      state.isDirty = true;
    }),

    // ... 其他方法实现
  }))
);
```

```typescript
export interface DataTypeDefinition {
  type: string;                  // 类型名称
  color: string;                 // Tailwind 颜色类

  noDefault?: boolean;           // 是否禁用默认值
```

```typescript
// 按分类获取类型
export function getTypesByCategory(database: DatabaseDialect): Record<string, string[]> {
  const types = dbToTypes[database];
  if (!types) return {};

  const categories: Record<string, string[]> = {
    '整数': [],
    '小数': [],
    '字符串': [],
    '日期时间': [],
    '布尔': [],
    '二进制': [],
    'JSON': [],
    '其他': [],
  };

  for (const [name, def] of Object.entries(types)) {
    if (def.color.includes('yellow')) categories['整数'].push(name);
    else if (def.color.includes('lime')) categories['小数'].push(name);
    else if (def.color.includes('orange')) categories['字符串'].push(name);
    else if (def.color.includes('cyan')) categories['日期时间'].push(name);
    else if (def.color.includes('violet')) categories['布尔'].push(name);
    else if (def.color.includes('emerald')) categories['二进制'].push(name);
    else if (def.color.includes('indigo')) categories['JSON'].push(name);
    else categories['其他'].push(name);
  }

  return categories;
}
```

```typescript
  quoteIdentifier: (name: string) => string;
```

```typescript
      </td>

      {/* 默认值 - 带验证 */}
      <td>
```

```typescript
// front/utils/sql-generator/shared.ts

import { TableField, TableNode, DiagramData, DatabaseDialect } from '../../components/database/designer/types';
import { getTypeDefinition } from '../../data/datatypes';

// 判断是否为 SQL 函数 (如 NOW(), CURRENT_TIMESTAMP)
const SQL_FUNCTIONS = [
  'NOW()', 'CURRENT_TIMESTAMP', 'CURRENT_DATE', 'CURRENT_TIME',
  'GETDATE()', 'SYSDATE', 'UUID()', 'GEN_RANDOM_UUID()',
];

export function isFunction(value: string): boolean {
  return SQL_FUNCTIONS.some(fn =>
    value.toUpperCase().includes(fn.toUpperCase())
  );
}

// 判断是否为 SQL 关键字
const SQL_KEYWORDS = ['NULL', 'TRUE', 'FALSE', 'DEFAULT'];

export function isKeyword(value: string): boolean {
  return SQL_KEYWORDS.includes(value.toUpperCase());
}

// 转义单引号
export function escapeQuotes(str: string): string {
  return str.replace(/'/g, "''");
}

// 解析默认值 (根据是否需要引号)
export function parseDefault(field: TableField, database: DatabaseDialect): string {
  if (!field.default || field.default === '') return '';

  // 函数和关键字不需要引号
  if (isFunction(field.default) || isKeyword(field.default)) {
    return field.default;
  }

  const typeDef = getTypeDefinition(database, field.type);
  if (typeDef && !typeDef.hasQuotes) {
    return field.default;
  }

  return `'${escapeQuotes(field.default)}'`;
}

// 生成字段注释 (多行)
export function exportFieldComment(comment: string): string {
  if (!comment || comment === '') return '';

  return comment
    .split('\n')
    .map(line => `\t-- ${line}\n`)
    .join('');
}

// 生成内联外键约束 (SQLite 需要)
export function getInlineFK(table: TableNode, diagram: DiagramData): string {
  const fks: string[] = [];

  diagram.relationships.forEach(r => {
    if (r.startTableId === table.id) {
      const startField = table.fields.find(f => f.id === r.startFieldId);
      const endTable = diagram.tables.find(t => t.id === r.endTableId);
      const endField = endTable?.fields.find(f => f.id === r.endFieldId);

      if (startField && endTable && endField) {
        fks.push(
          `\tFOREIGN KEY ("${startField.name}") REFERENCES "${endTable.name}"("${endField.name}")\n` +
          `\tON UPDATE ${r.updateConstraint} ON DELETE ${r.deleteConstraint}`
        );
      }
    }
  });

  return fks.join(',\n');
}

// 解析类型大小
export function parseTypeSize(field: TableField, database: DatabaseDialect): string {
  const typeDef = getTypeDefinition(database, field.type);
  if (!typeDef) return field.type;

  let result = field.type;

  // ENUM/SET 特殊处理
  if (field.type === 'ENUM' || field.type === 'SET') {
    if (field.values && field.values.length > 0) {
      result += `(${field.values.map(v => `'${v}'`).join(', ')})`;
    }
    return result;
  }

  // 有大小或精度的类型
  if ((typeDef.isSized || typeDef.hasPrecision) && field.size) {
    result += `(${field.size})`;
  }

  return result;
}
```

### 6.3 MySQL SQL 生成器

```typescript
// front/utils/sql-generator/mysql.ts

import { DiagramData, TableNode, TableField } from '../../components/database/designer/types';
import { escapeQuotes, parseDefault, parseTypeSize } from './shared';
import { getTypeDefinition } from '../../data/datatypes';

export function toMySQL(diagram: DiagramData): string {
  const tableStatements = diagram.tables.map(table => generateTableSQL(table, diagram)).join('\n\n');
  const fkStatements = generateForeignKeys(diagram);

  return `${tableStatements}\n\n${fkStatements}`;
}

function generateTableSQL(table: TableNode, diagram: DiagramData): string {
  // 生成字段定义
  const fieldDefs = table.fields.map(field => {
    const typeDef = getTypeDefinition('mysql', field.type);

    let def = `\t\`${field.name}\` ${parseTypeSize(field, 'mysql')}`;

    // UNSIGNED
    if (typeDef?.signed && field.unsigned) {
      def += ' UNSIGNED';
    }

    // NOT NULL
    if (field.notNull) {
      def += ' NOT NULL';
    }

    // AUTO_INCREMENT
    if (field.increment) {
      def += ' AUTO_INCREMENT';
    }

    // UNIQUE
    if (field.unique) {
      def += ' UNIQUE';
    }

    // DEFAULT
    if (field.default !== '') {
      def += ` DEFAULT ${parseDefault(field, 'mysql')}`;
    }

    // CHECK
    if (field.check !== '' && typeDef?.hasCheck) {
      def += ` CHECK(${field.check})`;
    }

    // COMMENT
    if (field.comment) {
      def += ` COMMENT '${escapeQuotes(field.comment)}'`;
    }

    return def;
  }).join(',\n');

  // 生成主键
  const primaryKeys = table.fields.filter(f => f.primary);
  const pkDef = primaryKeys.length > 0
    ? `,\n\tPRIMARY KEY(${primaryKeys.map(f => `\`${f.name}\``).join(', ')})`
    : '';

  // 生成索引
  const indexDefs = table.indices.map(idx =>
    `\nCREATE ${idx.unique ? 'UNIQUE ' : ''}INDEX \`${idx.name}\`\n` +
    `ON \`${table.name}\` (${idx.fields.map(f => `\`${f}\``).join(', ')});`
  ).join('');

  // 表注释
  const tableComment = table.comment
    ? ` COMMENT='${escapeQuotes(table.comment)}'`
    : '';

  return `CREATE TABLE IF NOT EXISTS \`${table.name}\` (\n${fieldDefs}${pkDef}\n)${tableComment};\n${indexDefs}`;
}

function generateForeignKeys(diagram: DiagramData): string {
  return diagram.relationships.map(r => {
    const startTable = diagram.tables.find(t => t.id === r.startTableId);
    const endTable = diagram.tables.find(t => t.id === r.endTableId);

    if (!startTable || !endTable) return '';

    const startField = startTable.fields.find(f => f.id === r.startFieldId);
    const endField = endTable.fields.find(f => f.id === r.endFieldId);

    if (!startField || !endField) return '';

    return `ALTER TABLE \`${startTable.name}\`\n` +
      `ADD FOREIGN KEY(\`${startField.name}\`) REFERENCES \`${endTable.name}\`(\`${endField.name}\`)\n` +
      `ON UPDATE ${r.updateConstraint} ON DELETE ${r.deleteConstraint};`;
  }).filter(Boolean).join('\n');
}
```

```typescript
// front/utils/sql-generator/postgresql.ts

import { DiagramData, TableNode } from '../../components/database/designer/types';
import { escapeQuotes, parseDefault, exportFieldComment, parseTypeSize } from './shared';
import { getTypeDefinition } from '../../data/datatypes';

export function toPostgres(diagram: DiagramData): string {
  // 生成 ENUM 定义
  const enumStatements = (diagram.enums || []).map(e =>
    `CREATE TYPE "${e.name}" AS ENUM (\n${e.values.map(v => `\t'${v}'`).join(',\n')}\n);`
  ).join('\n\n');

  // 生成自定义类型
  const typeStatements = (diagram.types || []).map(type => {
    const typeDef = `CREATE TYPE ${type.name} AS (\n${type.fields.map(f =>
      `\t${f.name} ${f.type}`
    ).join(',\n')}\n);`;

    const comment = type.comment?.trim()
      ? `\nCOMMENT ON TYPE "${type.name}" IS '${escapeQuotes(type.comment)}';`
      : '';

    return typeDef + comment;
  }).join('\n\n');

  // 生成表定义
  const tableStatements = diagram.tables.map(table =>
    generateTableSQL(table, diagram)
  ).join('\n\n');

  // 生成外键
  const fkStatements = generateForeignKeys(diagram);

  return [enumStatements, typeStatements, tableStatements, fkStatements]
    .filter(Boolean)
    .join('\n\n');
}

function generateTableSQL(table: TableNode, diagram: DiagramData): string {
  // 字段定义
  const fieldDefs = table.fields.map(field => {
    const typeDef = getTypeDefinition('postgresql', field.type);

    let def = exportFieldComment(field.comment);
    def += `\t"${field.name}" ${parseTypeSize(field, 'postgresql')}`;

    // ARRAY
    if (field.isArray) {
      def += ' ARRAY';
    }

    // NOT NULL
    if (field.notNull) {
      def += ' NOT NULL';
    }

    // UNIQUE
    if (field.unique) {
      def += ' UNIQUE';
    }

    // GENERATED AS IDENTITY (替代 SERIAL)
    if (field.increment) {
      def += ' GENERATED BY DEFAULT AS IDENTITY';
    }

    // DEFAULT
    if (field.default?.trim()) {
      def += ` DEFAULT ${parseDefault(field, 'postgresql')}`;
    }

    // CHECK
    if (field.check && typeDef?.hasCheck) {
      def += ` CHECK(${field.check})`;
    }

    return def;
  }).join(',\n');

  // 主键
  const primaryKeys = table.fields.filter(f => f.primary);
  const pkDef = primaryKeys.length > 0
    ? `,\n\tPRIMARY KEY(${primaryKeys.map(f => `"${f.name}"`).join(', ')})`
    : '';

  // 继承 (PostgreSQL 特有)
  const inheritsClause = table.inherits && table.inherits.length > 0
    ? `\n) INHERITS (${table.inherits.map(p => `"${p}"`).join(', ')})`
    : '\n)';

  // 注释
  const comments: string[] = [];
  if (table.comment?.trim()) {
    comments.push(`COMMENT ON TABLE "${table.name}" IS '${escapeQuotes(table.comment)}';`);
  }
  table.fields.forEach(f => {
    if (f.comment?.trim()) {
      comments.push(`COMMENT ON COLUMN "${table.name}"."${f.name}" IS '${escapeQuotes(f.comment)}';`);
    }
  });

  // 索引
  const indexDefs = table.indices.map(idx =>
    `CREATE ${idx.unique ? 'UNIQUE ' : ''}INDEX "${idx.name}"\n` +
    `ON "${table.name}" (${idx.fields.map(f => `"${f}"`).join(', ')});`
  ).join('\n');

  return `CREATE TABLE IF NOT EXISTS "${table.name}" (\n${fieldDefs}${pkDef}${inheritsClause};\n\n${comments.join('\n')}\n${indexDefs}`;
}

function generateForeignKeys(diagram: DiagramData): string {
  return diagram.relationships.map(r => {
    const startTable = diagram.tables.find(t => t.id === r.startTableId);
    const endTable = diagram.tables.find(t => t.id === r.endTableId);
    const startField = startTable?.fields.find(f => f.id === r.startFieldId);
    const endField = endTable?.fields.find(f => f.id === r.endFieldId);

    if (!startTable || !endTable || !startField || !endField) return '';

    return `ALTER TABLE "${startTable.name}"\n` +
      `ADD FOREIGN KEY("${startField.name}") REFERENCES "${endTable.name}"("${endField.name}")\n` +
      `ON UPDATE ${r.updateConstraint} ON DELETE ${r.deleteConstraint};`;
  }).filter(Boolean).join('\n');
}
```

```typescript
  const escapedTable = escapeQuotes(tableName);

    return `
EXEC sys.sp_addextendedproperty

    @level1type=N'TABLE',@level1name=N'${escapedTable}',
    @level2type=N'COLUMN',@level2name=N'${escapedColumn}';
GO
`;

    : '';
```

```typescript
import { parseDefault, parseTypeSize } from './shared';
import { getTypeDefinition } from '../../data/datatypes';

export function toOracle(diagram: DiagramData): string {
  const tableStatements = diagram.tables.map(table =>
    generateTableSQL(table, diagram)
  ).join('\n\n');

  const fkStatements = generateForeignKeys(diagram);

  return `${tableStatements}\n\n${fkStatements}`;
}

function generateTableSQL(table: TableNode, diagram: DiagramData): string {
  // 表注释 (Oracle 使用 /* */ 形式)
  const tableComment = table.comment ? `/* ${table.comment} */\n` : '';

  const fieldDefs = table.fields.map(field => {
    const typeDef = getTypeDefinition('oracle', field.type);

    // 字段注释
    let def = field.comment ? `\t-- ${field.comment}\n` : '';
    def += `\t"${field.name}" ${parseTypeSize(field, 'oracle')}`;

    if (field.notNull) def += ' NOT NULL';

    // Oracle 使用 GENERATED ALWAYS AS IDENTITY
    if (field.increment) {
      def += ' GENERATED ALWAYS AS IDENTITY';
    }

    if (field.unique) def += ' UNIQUE';

    if (field.default !== '') {
      def += ` DEFAULT ${parseDefault(field, 'oracle')}`;
    }

    if (field.check !== '' && typeDef?.hasCheck) {
      def += ` CHECK(${field.check})`;
    }

    return def;
  }).join(',\n');

  // 主键
  const primaryKeys = table.fields.filter(f => f.primary);
  const pkDef = primaryKeys.length > 0
    ? `,\n\tPRIMARY KEY(${primaryKeys.map(f => `"${f.name}"`).join(', ')})`
    : '';

  // 索引
  const indexDefs = table.indices.map(idx =>
    `\nCREATE ${idx.unique ? 'UNIQUE ' : ''}INDEX "${idx.name}"\n` +
    `ON "${table.name}" (${idx.fields.map(f => `"${f}"`).join(', ')});`
  ).join('');

  const tableCommentSuffix = table.comment ? ` -- ${table.comment}` : '';

  return `${tableComment}CREATE TABLE "${table.name}" (\n${fieldDefs}${pkDef}\n)${tableCommentSuffix};\n${indexDefs}`;
}

function generateForeignKeys(diagram: DiagramData): string {
  return diagram.relationships.map(r => {
    const startTable = diagram.tables.find(t => t.id === r.startTableId);
    const endTable = diagram.tables.find(t => t.id === r.endTableId);
    const startField = startTable?.fields.find(f => f.id === r.startFieldId);
    const endField = endTable?.fields.find(f => f.id === r.endFieldId);

    if (!startTable || !endTable || !startField || !endField) return '';

    // Oracle 外键需要约束名
    return `ALTER TABLE "${startTable.name}"\n` +
      `ADD CONSTRAINT "${r.name}" FOREIGN KEY ("${startField.name}") ` +
      `REFERENCES "${endTable.name}" ("${endField.name}")\n` +
      `ON UPDATE ${r.updateConstraint} ON DELETE ${r.deleteConstraint};`;
  }).filter(Boolean).join('\n');
}
```

```rust
impl DatabaseService {

                ).await?)

                    &request.host, request.port, &request.username,
                    &request.password, request.database.as_deref()
                ).await?)
            }

                Arc::new(SqliteDriver::connect(&request.file_path.unwrap_or_default()).await?)
            }
            DatabaseType::MSSQL => {
                Arc::new(MssqlDriver::connect(
                    &request.host, request.port, &request.username,
                    &request.password, request.database.as_deref()
                ).await?)
            }
            DatabaseType::Oracle => {
                Arc::new(OracleDriver::connect(
                    &request.host, request.port, &request.username,
                    &request.password, &request.service_name.unwrap_or_default()
                ).await?)
            }
        };

            request.db_type,
            request.host.clone(),
            request.port,
            request.database,
            None,
            driver,
        ));

        Ok(ConnectionInfo { connection_id: request.connection_id })
```

### A.1 SQL 语法差异

```sql
-- MySQL/MariaDB
CREATE TABLE IF NOT EXISTS `users` (
    `id` INT AUTO_INCREMENT,
    `name` VARCHAR(100) NOT NULL COMMENT 'User name',
    PRIMARY KEY(`id`)

-- PostgreSQL
CREATE TABLE IF NOT EXISTS "users" (
    "id" SERIAL,
    "name" VARCHAR(100) NOT NULL,
    PRIMARY KEY("id")
);
COMMENT ON TABLE "users" IS 'Users table';
COMMENT ON COLUMN "users"."name" IS 'User name';

-- SQLite
CREATE TABLE IF NOT EXISTS "users" (
    "id" INTEGER PRIMARY KEY,  -- 自动自增
    "name" TEXT NOT NULL
);

-- MSSQL
CREATE TABLE [users] (
    [id] INT IDENTITY,
    [name] VARCHAR(100) NOT NULL,
    PRIMARY KEY([id])
);
GO
EXEC sp_addextendedproperty
    @name='MS_Description', @value='Users table',
    @level0type='SCHEMA', @level0name='dbo',
    @level1type='TABLE', @level1name='users';
GO

-- Oracle
CREATE TABLE "users" (
    "id" NUMBER GENERATED ALWAYS AS IDENTITY,
    "name" VARCHAR2(100) NOT NULL,
    PRIMARY KEY("id")
);
```
