## 2. SQL Server 与 MySQL 主要差异

### 2.1 数据类型映射

| MySQL 类型 | SQL Server 类型 | 说明 |
|-----------|----------------|------|
| INT | INT | 4 字节整数 |
| BIGINT | BIGINT | 8 字节整数 |
| SMALLINT | SMALLINT | 2 字节整数 |
| TINYINT | TINYINT | 1 字节（SQL Server: 0-255 无符号） |
| DECIMAL(p,s) | DECIMAL(p,s) | 精确小数 |
| FLOAT | FLOAT | 近似数值 |
| DOUBLE | FLOAT(53) / REAL | SQL Server 无 DOUBLE |
| VARCHAR(n) | VARCHAR(n) | 可变长字符串（最大 8000） |
| VARCHAR(MAX) | VARCHAR(MAX) | 可变长字符串（最大 2GB） |
| NVARCHAR(n) | NVARCHAR(n) | Unicode 可变字符串 |
| TEXT | VARCHAR(MAX) | TEXT 已废弃 |
| LONGTEXT | VARCHAR(MAX) | 使用 MAX |
| BLOB | VARBINARY(MAX) | 二进制大对象 |
| DATETIME | DATETIME2 | 推荐使用 DATETIME2 |
| TIMESTAMP | ROWVERSION | SQL Server TIMESTAMP 是行版本 |
| DATE | DATE | 仅日期 |
| TIME | TIME | 仅时间 |
| BOOLEAN | BIT | 0/1 |
| JSON | NVARCHAR(MAX) | 配合 JSON 函数使用 |
| UUID | UNIQUEIDENTIFIER | 全局唯一标识符 |

### 2.2 SQL Server 特有类型

| 类型 | 说明 |
|------|------|
| MONEY | 货币类型，8 字节 |
| SMALLMONEY | 小货币类型，4 字节 |
| DATETIME2(n) | 扩展日期时间，精度 0-7 |
| DATETIMEOFFSET(n) | 带时区日期时间 |
| UNIQUEIDENTIFIER | 16 字节 GUID |
| XML | XML 数据 |
| GEOGRAPHY | 地理空间数据 |
| GEOMETRY | 几何空间数据 |
| HIERARCHYID | 层次结构数据 |
| SQL_VARIANT | 可存储多种数据类型 |
| ROWVERSION | 自动版本戳（原 TIMESTAMP） |

### 2.3 语法差异

#### 2.3.1 标识符引用
```sql
-- MySQL: 反引号
`table_name`

-- SQL Server: 方括号或双引号
[table_name]
"table_name"
```

#### 2.3.2 自增主键
```sql
-- MySQL
id INT AUTO_INCREMENT PRIMARY KEY

-- SQL Server (IDENTITY)
id INT IDENTITY(1,1) PRIMARY KEY
-- IDENTITY(seed, increment)
```

#### 2.3.3 注释语法
```sql
-- MySQL: 内联注释
CREATE TABLE users (
  id INT COMMENT '用户ID'
) COMMENT='用户表';

-- SQL Server: 扩展属性
EXEC sp_addextendedproperty
  @name = N'MS_Description',
  @value = N'用户表',
  @level0type = N'SCHEMA', @level0name = N'dbo',
  @level1type = N'TABLE', @level1name = N'users';

EXEC sp_addextendedproperty
  @name = N'MS_Description',
  @value = N'用户ID',
  @level0type = N'SCHEMA', @level0name = N'dbo',
  @level1type = N'TABLE', @level1name = N'users',
  @level2type = N'COLUMN', @level2name = N'id';
```

#### 2.3.4 分页
```sql
-- MySQL
SELECT * FROM table LIMIT 10 OFFSET 20;

-- SQL Server 2012+
SELECT * FROM table
ORDER BY id
OFFSET 20 ROWS
FETCH NEXT 10 ROWS ONLY;

-- SQL Server 2008 及以下
SELECT TOP 10 * FROM (
  SELECT *, ROW_NUMBER() OVER (ORDER BY id) AS rn
  FROM table
) t WHERE rn > 20;
```

#### 2.3.5 获取插入的 ID
```sql
-- MySQL
SELECT LAST_INSERT_ID();

-- SQL Server
SELECT SCOPE_IDENTITY();
-- 或
SELECT @@IDENTITY;
-- 或使用 OUTPUT 子句
INSERT INTO table (name) OUTPUT inserted.id VALUES ('test');
```

#### 2.3.6 批处理分隔符
```sql
-- SQL Server 使用 GO 分隔批处理
CREATE TABLE [users] (...);
GO

CREATE TABLE [orders] (...);
GO
```

//! SQL Server database driver implementation

use std::collections::HashMap;
use std::time::Instant;

use async_trait::async_trait;
use tiberius::{Client, Config, AuthMethod, ColumnData};
use tokio::net::TcpStream;
use tokio_util::compat::TokioAsyncWriteCompatExt;

    CheckConstraintInfo, DatabaseObjectsCount, ForeignKeyInfo, QueryColumn, QueryResult,
    RoutineInfo, TableInfo, TableOptions, TableStructure, TriggerInfo, ViewInfo,

/// SQL Server database driver
pub struct MssqlDriver {
    client: Client<tokio_util::compat::Compat<TcpStream>>,
}

impl MssqlDriver {
    /// Create a new SQL Server connection
    pub async fn connect(
        host: &str,
        port: u16,
        username: &str,
        password: &str,
        database: &str,
    ) -> Result<Self, String> {
        let mut config = Config::new();
        config.host(host);
        config.port(port);
        config.authentication(AuthMethod::sql_server(username, password));
        config.database(database);
        config.trust_cert(); // 开发环境信任自签名证书

        let tcp = TcpStream::connect(config.get_addr())
            .await
            .map_err(|e| format!("TCP connection failed: {}", e))?;

        let client = Client::connect(config, tcp.compat_write())
            .await
            .map_err(|e| format!("SQL Server connection failed: {}", e))?;

        log::info!("SQL Server connection established: {}:{}/{}", host, port, database);
        Ok(Self { client })
    }

    /// Test connection without keeping it open
    pub async fn test_connection(
        host: &str,
        port: u16,
        username: &str,
        password: &str,
        database: &str,
    ) -> Result<(), String> {
        let driver = Self::connect(host, port, username, password, database).await?;
        driver.close().await;
        Ok(())
    }

    fn get_column_value(&self, col: &ColumnData) -> serde_json::Value {
        match col {
            ColumnData::I32(Some(v)) => serde_json::Value::from(*v),
            ColumnData::I64(Some(v)) => serde_json::Value::from(*v),
            ColumnData::I16(Some(v)) => serde_json::Value::from(*v),

            ColumnData::Numeric(Some(v)) => serde_json::Value::from(v.value().to_string()),
            _ => serde_json::Value::Null,
        }
    }
}

## 7. SQL 生成器实现

### 7.1 DDL 生成器

**文件**: `front/utils/sql-generator/mssql.ts`

```typescript
import { DiagramData, TableNode } from '../../components/database/designer/types'
import { escapeQuotes, parseDefault, parseTypeSize } from './shared'
import { getDataType } from '../../config/datatypes'

export function toMSSQL(diagram: DiagramData): string {
  const tableStatements = diagram.tables.map(table =>
    generateTableSQL(table, diagram)
  ).join('\n\n')

  const fkStatements = generateForeignKeys(diagram)

  return `${tableStatements}\n\n${fkStatements}`
}

// 生成扩展属性 SQL (用于注释)
function generateExtendedProperty(
  value: string,
  tableName: string,
  schemaName: string = 'dbo',
  columnName?: string
): string {
  if (!value || value.trim() === '') return ''

  const escapedValue = escapeQuotes(value.replace(/\n/g, ' '))

  if (columnName) {
    return `
EXEC sys.sp_addextendedproperty
    @name=N'MS_Description', @value=N'${escapedValue}',
    @level0type=N'SCHEMA', @level0name=N'${schemaName}',
    @level1type=N'TABLE', @level1name=N'${tableName}',
    @level2type=N'COLUMN', @level2name=N'${columnName}';
GO
`
  } else {
    return `
EXEC sys.sp_addextendedproperty
    @name=N'MS_Description', @value=N'${escapedValue}',
    @level0type=N'SCHEMA', @level0name=N'${schemaName}',
    @level1type=N'TABLE', @level1name=N'${tableName}';
GO
`
  }
}

function generateTableSQL(table: TableNode, diagram: DiagramData): string {
  const schemaName = 'dbo' // 可从 diagram 中获取

  const fieldDefs = table.fields.map(field => {
    const typeDef = getDataType('mssql', field.type)
    const isSized = typeDef?.isSized || typeDef?.hasPrecision

    let def = `\t[${field.name}] ${field.type}`

    // 类型大小
    if (field.size && isSized) {
      def += `(${field.size})`
    }

    // NOT NULL
    if (field.notNull) def += ' NOT NULL'

    // IDENTITY
    if (field.increment) def += ' IDENTITY(1,1)'

    // UNIQUE
    if (field.unique) def += ' UNIQUE'

    // DEFAULT
    if (field.default !== '') {
      def += ` DEFAULT ${parseDefault(field, 'mssql')}`
    }

    // CHECK
    if (field.check !== '' && typeDef?.hasCheck) {
      def += ` CHECK(${field.check})`
    }

    return def
  }).join(',\n')

  // 主键
  const primaryKeys = table.fields.filter(f => f.primary)
  const pkDef = primaryKeys.length > 0
    ? `,\n\tCONSTRAINT [PK_${table.name}] PRIMARY KEY CLUSTERED (${primaryKeys.map(f => `[${f.name}]`).join(', ')})`
    : ''

  const createTableSQL = `CREATE TABLE [${schemaName}].[${table.name}] (\n${fieldDefs}${pkDef}\n);\nGO\n`

  // 表注释
  const tableCommentSQL = generateExtendedProperty(table.comment, table.name, schemaName)

  // 列注释
  const columnCommentsSQL = table.fields
    .map(f => generateExtendedProperty(f.comment, table.name, schemaName, f.name))
    .join('')

  // 索引
  const indexDefs = table.indices.map(idx => {
    const indexType = idx.unique ? 'UNIQUE NONCLUSTERED' : 'NONCLUSTERED'
    return `
CREATE ${indexType} INDEX [${idx.name}]
ON [${schemaName}].[${table.name}] (${idx.fields.map(f => `[${f}]`).join(', ')});
GO
`
  }).join('')

  return `${createTableSQL}${tableCommentSQL}${columnCommentsSQL}${indexDefs}`
}

function generateForeignKeys(diagram: DiagramData): string {
  return diagram.relationships.map(r => {
    const startTable = diagram.tables.find(t => t.id === r.startTableId)
    const endTable = diagram.tables.find(t => t.id === r.endTableId)
    const startField = startTable?.fields.find(f => f.id === r.startFieldId)
    const endField = endTable?.fields.find(f => f.id === r.endFieldId)

    if (!startTable || !endTable || !startField || !endField) return ''

    const constraintName = r.name || `FK_${startTable.name}_${endTable.name}`

    return `
ALTER TABLE [dbo].[${startTable.name}]
ADD CONSTRAINT [${constraintName}]
FOREIGN KEY ([${startField.name}])
REFERENCES [dbo].[${endTable.name}] ([${endField.name}])
ON UPDATE ${r.updateConstraint} ON DELETE ${r.deleteConstraint};
GO`
  }).filter(Boolean).join('')
}
```

## 9. 特殊处理

### 9.1 IDENTITY 列

```sql
-- 创建表时
[id] INT IDENTITY(1,1) NOT NULL

-- 允许插入 IDENTITY 值
SET IDENTITY_INSERT [table_name] ON
INSERT INTO [table_name] ([id], [name]) VALUES (100, 'test')
SET IDENTITY_INSERT [table_name] OFF

-- 重置 IDENTITY 种子
DBCC CHECKIDENT ('[table_name]', RESEED, 0)
```

## 10. 限制说明

1. **IF NOT EXISTS**: SQL Server 2016 前不支持 `CREATE TABLE IF NOT EXISTS`，需使用：
   ```sql
   IF NOT EXISTS (SELECT * FROM sys.tables WHERE name = 'table_name')
   BEGIN
       CREATE TABLE [table_name] (...)
   END
   ```

2. **注释**: 不支持内联 COMMENT 语法，需使用 sp_addextendedproperty

3. **GO 分隔符**: GO 不是 SQL 语句，而是 SSMS/sqlcmd 的批处理分隔符

4. **TIMESTAMP**: SQL Server 的 TIMESTAMP 是 ROWVERSION（自动版本戳），不是日期时间

5. **LIMIT/OFFSET**: 不支持，使用 OFFSET FETCH 或 TOP + ROW_NUMBER()

6. **UNSIGNED**: 不支持无符号整数类型
